string ConnectString =

static void AddClient(NpgsqlConnection connection)
{
    string sql = """
    INSERT INTO clients
    (first_name, last_name, birth_date, phone, email,
    registration_date, status)
    VALUES
    (@firstName, @lastName, @birthDate, @phone, @email, @regDate, @status);
    """;

    using var command = new NpgsqlCommand(sql, connection);

    Console.WriteLine("Введите имя");
    string rata = Console.ReadLine();

    Console.WriteLine("Введите фамилию");
    string rataq = Console.ReadLine();

    Console.WriteLine("Введите телефон");
    string rataw = Console.ReadLine();

    string pattern = @"^(?:\+7|8)\d{10}$";

    if (!Regex.IsMatch(rataw, pattern))
    {
        Console.WriteLine("Неверный номер!");
        return;
    }

    command.Parameters.AddWithValue("phone", rataw);

    Console.WriteLine("Введите почту");
    string ratae = Console.ReadLine();

    Console.WriteLine("Введите дату рождения в формате (год.месяц.день)");
    string ratay = Console.ReadLine();

    try
    {
        DateTime asd = Convert.ToDateTime(ratay);
        command.Parameters.AddWithValue("birthDate", asd);
    }
    catch
    {
        Console.WriteLine("Введите правильное число");
        return;
    }

    command.Parameters.AddWithValue("firstName", rata);
    command.Parameters.AddWithValue("lastName", rataq);
    command.Parameters.AddWithValue("email", ratae);
    command.Parameters.AddWithValue("regDate", DateTime.Today);
    command.Parameters.AddWithValue("status", "active");

    int affectedRows = command.ExecuteNonQuery();

    Console.WriteLine($"Добавлено строк: {affectedRows}");
}

static void DeleteClient(NpgsqlConnection connection)
{
    try
    {
        string sql = """
        DELETE FROM clients
        WHERE client_id = @clientID;
        """;

        Console.WriteLine("Введите идентификатор, который хотите удалить");
        string ra = Console.ReadLine();

        if (!int.TryParse(ra, out int clientId))
        {
            Console.WriteLine("ID должен быть числом.");
            return;
        }

        using var command = new NpgsqlCommand(sql, connection);

        command.Parameters.AddWithValue("clientID", clientId);

        int affedtedRows = command.ExecuteNonQuery();

        if (affedtedRows == 0)
        {
            Console.WriteLine("Клиент не найден.");
        }
        else
        {
            Console.WriteLine("Клиент удалён.");
        }
    }
    catch (Exception ex)
    {
        Console.WriteLine(ex.ToString());
    }
}

static void UpdateClient(NpgsqlConnection connection)
{
    Console.WriteLine("Введите 1 или 2, чтобы выбрать что изменить телефон или почту");

    string ra = Console.ReadLine();

    if (!int.TryParse(ra, out int ta))
    {
        Console.WriteLine("ID должен быть числом.");
        return;
    }

    if (ta == 1)
    {
        Console.WriteLine("Введите айди");
        string rat = Console.ReadLine();

        int tat = int.Parse(rat);

        Console.WriteLine("Введите телефон");
        string rata = Console.ReadLine();

        string sql = """
        UPDATE clients 
        SET phone = @phone
        WHERE client_id = @clientId
        """;

        using var command = new NpgsqlCommand(sql, connection);

        string pattern = @"^(?:\+7|8)\d{10}$";

        if (!Regex.IsMatch(rata, pattern))
        {
            Console.WriteLine("Неверный номер!");
            return;
        }

        command.Parameters.AddWithValue("phone", rata);
        command.Parameters.AddWithValue("clientId", tat);

        int affectedRows = command.ExecuteNonQuery();

    }
    else if (ta == 2)
    {
        Console.WriteLine("Введите айди");
        string rat = Console.ReadLine();

        int tato = int.Parse(rat);

        Console.WriteLine("Введите почту");
        string ratar = Console.ReadLine();

        string sql = """
        UPDATE clients 
        SET email = @email
        WHERE client_id = @clientId
        """;

        using var command = new NpgsqlCommand(sql, connection);

        command.Parameters.AddWithValue("email", ratar);
        command.Parameters.AddWithValue("clientID", tato);

        int affectedRown = command.ExecuteNonQuery();

        Console.WriteLine($"Изменено строк: {affectedRown}");
    }
    else
    {
        Console.WriteLine("Введите правильное число");
    }
}

static void ShowClients(NpgsqlConnection connection)
{
    string sql = """
    SELECT client_id, first_name, last_name
    FROM clients
    WHERE status = @status AND birth_date < @birthDate;
    """;

    using var command = new NpgsqlCommand(sql, connection);

    command.Parameters.AddWithValue("status", "active");
    command.Parameters.AddWithValue("birthDate", new DateTime(2005, 10, 7));

    using var reader = command.ExecuteReader();

    while (reader.Read())
    {
        Console.WriteLine(
            $"{reader.GetInt32(0)} " +
            $"{reader.GetString(1)} " +
            $"{reader.GetString(2)}");
    }
}
